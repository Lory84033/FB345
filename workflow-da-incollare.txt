name: Compila APK

on:
  push:
    branches: [ main, master ]
  workflow_dispatch:

permissions:
  contents: write

jobs:
  build:
    runs-on: ubuntu-latest
    steps:
      - name: Scarica il codice
        uses: actions/checkout@v4

      - name: Installa Java 17
        uses: actions/setup-java@v4
        with:
          distribution: temurin
          java-version: '17'

      - name: Prepara Android SDK
        uses: android-actions/setup-android@v3

      - name: Componenti Android
        run: sdkmanager "platforms;android-34" "build-tools;34.0.0" || true

      - name: Compila APK
        run: |
          chmod +x gradlew
          ./gradlew --no-daemon assembleRelease

      - name: Rinomina il file
        run: |
          mkdir -p out
          cp app/build/outputs/apk/release/app-release.apk out/Friendbook.apk

      - name: Salva come allegato
        uses: actions/upload-artifact@v4
        with:
          name: Friendbook-apk
          path: out/Friendbook.apk

      - name: Pubblica in Releases
        uses: softprops/action-gh-release@v2
        with:
          tag_name: apk-latest
          name: Friendbook (ultima versione)
          body: Scarica Friendbook.apk e installalo sul telefono.
          files: out/Friendbook.apk
          make_latest: 'true'
